import { chromium } from '@playwright/test';
const BASE = 'http://localhost:4200';
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium', args: ['--no-sandbox'] });
const context = await browser.newContext({ viewport: { width: 1280, height: 800 }, locale: 'es-BO' });
const page = await context.newPage();
const malos = [];
const consola = [];
page.on('response', (r) => { if (r.status() >= 400) malos.push(`${r.status()} ${r.url()}`); });
page.on('console', (m) => { if (m.type() === 'warning' || m.type() === 'error') consola.push(`${m.type()} ${m.text().slice(0, 120)}`); });
const entrar = async () => {
  await page.goto(`${BASE}/auth`, { waitUntil: 'domcontentloaded', timeout: 120_000 });
  await page.getByTestId('login-identifier').fill('gobernanza@alovida.dev');
  await page.getByTestId('login-password').fill('dev');
  await page.getByTestId('login-submit').click();
  await page.waitForURL(/\/(dashboard|auth\/organization)/, { timeout: 60_000 });
  if (page.url().includes('/auth/organization')) { await page.getByTestId('tenant-opcion').first().click(); await page.waitForURL(/\/dashboard/, { timeout: 60_000 }); }
};
await entrar();
await page.waitForTimeout(1500);
console.log('--- dashboard 4xx:', [...new Set(malos)].join('\n'));
console.log('--- dashboard consola:', [...new Set(consola)].join('\n'));
malos.length = 0; consola.length = 0;
await page.goto(`${BASE}/administration/portal`, { waitUntil: 'domcontentloaded', timeout: 120_000 });
await page.getByTestId('portal-cifras').waitFor({ timeout: 60_000 });
await page.waitForTimeout(1500);
console.log('--- portal 4xx:', [...new Set(malos)].join('\n'));
console.log('--- portal consola:', [...new Set(consola)].join('\n'));
await browser.close();
